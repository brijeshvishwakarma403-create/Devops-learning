#!/usr/bin/env python3

# Importing Library

import requests

url = "https://jsonplaceholder.typicode.com/posts"

# Checking that url is responsive or not
# then only move forward.

first_part_success = False

# Printing Status Code.
try:
    response = requests.get(url, timeout=10)
    print("status_code: ", response.status_code)

    response.raise_for_status()

    posts = response.json()

# Using for loop to print 5 post data.

    for post in posts[:5]:
        print("Post ID:", post["id"])
        print("User ID:", post["userId"])
        print("Title:", post["title"])
        print("Body:", post["body"])
        print("---------------------")

# checking if it is successful or failed.
        first_part_success = True

except requests.exceptions.RequestException as err:
    print("Error:", err)


"""Additional Task using
if condition to move ahead only
if first part of code is executed successfully."""

if first_part_success:

    post_id = input("Enter the post id ")
    url2 = "https://jsonplaceholder.typicode.com/posts/" + post_id

    try:
        response2 = requests.get(url2, timeout=10)
        print("Status Code:", response2.status_code)

        response2.raise_for_status()

        post = response2.json()
        print("Post ID:", post["id"])
        print("User ID:", post["userId"])
        print("Title:", post["title"])
        print("Body:", post["body"])

    except requests.exceptions.RequestException as err:
        print("Error:", err)

print("------------------------------------------------------")

# Now let's try to post data in the table

response3 = requests.post("https://jsonplaceholder.typicode.com/posts",
    json={"title": "Brijesh", "body": "Testing post method", "userId": 1}
)
print("Status Code:", response3.status_code)
print(response3.text, "\n" + "POST Successful")

print("------------------------------------------------------")

# using PUT method to update the data

response3 = requests.put("https://jsonplaceholder.typicode.com/posts/11",
    json={"title": "UBrijesh", "body": "UTesting post method", "userId": 11})
print("Status Code:", response3.status_code)
print("PUT operation successful")

print("------------------------------------------------------")

# Using DELETE method to delete the data
response3 = requests.delete("https://jsonplaceholder.typicode.com/posts/1")
print("Status Code:", response3.status_code)
print("DELETE Successful")
print("------------------------------------------------------")
