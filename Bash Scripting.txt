Task 10: Monitor Disk Usage

root@JARVIS:~# cat d_usage.sh

#!/bin/bash

for DRIVE in /mnt/c /mnt/d /mnt/e
do
    usage=$(df -h "$DRIVE" |tail -1 | awk '{print $5}'| tr -d '%')
      echo "Drive $DRIVE Usage: $usage%"

    if [[ $usage -gt 80 ]]; then
      echo "WARNING: Disk usage is above 80%"
    else
      echo "Disk usage is normal"
    fi
done

root@JARVIS:~# cat health_checker.sh
#!/bin/bash
read -p "Enter service name to monitor: " SERVICE

echo "Monitoring started. Press Ctrl + c to Stop."
echo "---------------------------------------------"

while true
do
   for DRIVE in /mnt/c /mnt/d /mnt/e
   do
     ls "$DRIVE" &> /dev/null
     if [[ ! -d "$DRIVE" ]]; then
      echo "Drive $DRIVE not found - skipping"
      continue
     fi

   USAGE=$(df -h "$DRIVE" | tail -1 | awk '{print $5}' | tr -d '%')

     if [[ $USAGE -ge 80 ]]; then
      echo "CRITICAL: Disk usage is at (${USAGE}%)"
     else
      echo "Disk is normal (${USAGE}%)"
     fi
   done

   echo "---------------------------------------"

   if systemctl is-active --quiet "$SERVICE"; then
      echo "Service '$SERVICE' is running"
   else
      echo "Service '$SERVICE' is stopped"
   fi

   echo "---------------------------------------"

   for (( i=60; i>0; i--))
   do
   echo -ne "Next check in: $i seconds \r"
   sleep 1
   done
   echo -ne "\n"
done
